if(NOT TARGET PCSX2)
  message(FATAL_ERROR "Configure the pinned PCSX2 project with CMAKE_PROJECT_Pcsx2_INCLUDE=research/pcsx2_bridge/inject.cmake")
endif()
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
set(BRIDGE_SOURCE "${CMAKE_CURRENT_LIST_DIR}")
set(HOST_SHIM "${CMAKE_BINARY_DIR}/dw3_gs_bridge/public_host.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${BRIDGE_SOURCE}/generate_host.py"
  "${CMAKE_SOURCE_DIR}" "${HOST_SHIM}" COMMAND_ERROR_IS_FATAL ANY)
add_library(dw3_gs_bridge SHARED "${HOST_SHIM}")
target_compile_features(dw3_gs_bridge PRIVATE cxx_std_20)
target_compile_definitions(dw3_gs_bridge PRIVATE DW3_GS_BUILD)
if(MSVC)
  target_compile_options(dw3_gs_bridge PRIVATE /EHsc)
endif()
target_include_directories(dw3_gs_bridge PRIVATE "${BRIDGE_SOURCE}"
  "${CMAKE_BINARY_DIR}/common/include" "${CMAKE_SOURCE_DIR}/pcsx2")
target_link_libraries(dw3_gs_bridge PRIVATE PCSX2_FLAGS PCSX2)
add_executable(dw3_gs_probe "${BRIDGE_SOURCE}/probe.cpp")
target_compile_features(dw3_gs_probe PRIVATE cxx_std_20)
target_link_libraries(dw3_gs_probe PRIVATE dw3_gs_bridge)
if(MSVC)
  target_compile_options(dw3_gs_probe PRIVATE /EHsc)
endif()
add_executable(dw3_gs_contracts "${BRIDGE_SOURCE}/contracts.cpp")
target_compile_features(dw3_gs_contracts PRIVATE cxx_std_20)
target_link_libraries(dw3_gs_contracts PRIVATE dw3_gs_bridge)
if(MSVC)
  target_compile_options(dw3_gs_contracts PRIVATE /EHsc)
endif()
